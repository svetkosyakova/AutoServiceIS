# UML-диаграмма классов информационной системы «Автосервис»

```mermaid
classDiagram
direction LR

class Клиент {
    +int клиентId
    +string ФИО
    +string телефон
    +string email
    +добавитьАвтомобиль()
    +получитьИсториюОбслуживания()
    +изменитьКонтакты()
}

class Автомобиль {
    +int автомобильId
    +string VIN
    +string госномер
    +int годВыпуска
    +изменитьДанные()
    +получитьИсториюЗаказов()
}

class Марка {
    +int маркаId
    +string название
    +получитьСписокМоделей()
    +добавитьМодель()
    +изменитьДанные()
}

class Модель {
    +int модельId
    +string название
    +int маркаId
    +получитьИнформацию()
    +изменитьДанные()
}

class СТО {
    +int стоId
    +string название
    +string адрес
    +string телефон
    +получитьСписокЗаказов()
    +получитьСписокСотрудников()
    +изменитьДанные()
}

class ЗаказНаряд {
    +int заказНарядId
    +date датаСоздания
    +date датаЗавершения
    +string статус
    +decimal итоговаяСтоимость
    +добавитьРаботу()
    +добавитьЗапчасть()
    +назначитьМастера()
    +рассчитатьСтоимость()
    +изменитьСтатус()
    +закрытьЗаказ()
}

class Менеджер {
    +int менеджерId
    +оформитьЗаказ()
    +назначитьМастера()
    +изменитьСтатусЗаказа()
    +выдатьЗаказКлиенту()
}

class Сотрудник {
    +int сотрудникId
    +string ФИО
    +string телефон
    +string должность
    +изменитьКонтакты()
    +получитьИнформацию()
}

class Мастер {
    +int мастерId
    +string специализация
    +взятьЗаказВРаботу()
    +отметитьВыполненнуюРаботу()
    +завершитьРаботу()
    +получитьСписокЗаказов()
}

class Работа {
    +int работаId
    +string название
    +string описание
    +decimal базоваяСтоимость
    +получитьСтоимость()
    +изменитьСтоимость()
    +изменитьОписание()
}

class ВыполненнаяРабота {
    +int выполненнаяРаботаId
    +decimal стоимость
    +string результат
    +date датаВыполнения
    +зафиксироватьРезультат()
    +рассчитатьСтоимость()
    +изменитьРезультат()
}

class Запчасть {
    +int запчастьId
    +string название
    +string артикул
    +decimal цена
    +получитьИнформацию()
    +изменитьЦену()
    +проверитьДанные()
}

class ИспользованнаяЗапчасть {
    +int использованнаяЗапчастьId
    +int количество
    +decimal ценаНаМоментИспользования
    +рассчитатьСтоимость()
    +изменитьКоличество()
}

Клиент "1" --> "0..*" Автомобиль : владеет
Марка "1" --> "0..*" Модель : включает
Модель "1" --> "0..*" Автомобиль : определяет
СТО "1" --> "0..*" ЗаказНаряд : обслуживает
Автомобиль "1" --> "0..*" ЗаказНаряд : обслуживается
Клиент "1" --> "0..*" ЗаказНаряд : оформляет
Сотрудник <|-- Менеджер
Сотрудник <|-- Мастер
Менеджер "1" --> "0..*" ЗаказНаряд : оформляет
Мастер "1" --> "0..*" ЗаказНаряд : выполняет
ЗаказНаряд "1" *-- "0..*" ВыполненнаяРабота : содержит
Работа "1" --> "0..*" ВыполненнаяРабота : описывает
ЗаказНаряд "1" *-- "0..*" ИспользованнаяЗапчасть : содержит
Запчасть "1" --> "0..*" ИспользованнаяЗапчасть : используется
```

## Примечание

Диаграмма описывает основные классы, их атрибуты, операции и отношения. Состав атрибутов и методов является проектным вариантом и может уточняться при реализации системы.
